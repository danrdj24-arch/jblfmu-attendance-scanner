const APPS_SCRIPT_URL =

"https://script.google.com/macros/s/AKfycbyLzlhZxzBljBH8Pp_fxzH5hvMozzvCDPkhRoz3xcv4XVNvHpHgt05jsV_da7pHaAnq/exec";



const STATION_ID =

"JBL-MOLO-GATE-01";



const STATION_TOKEN =

"JBLMOLO-GATE-2026-8X7K";



/************************************************

 * DISPLAY RESULT

 ************************************************/


function showMessage(message, type) {


  const result =

    document.getElementById("result");


  result.innerHTML =

    '<div class="' + type + '">' +

    message +

    '</div>';


}



/************************************************

 * EXTRACT STUDENT ID

 ************************************************/


function extractStudentID(qrText) {


  qrText =

    String(qrText).trim();



  /*

   * If QR contains:

   *

   * https://.../exec?id=JBL-0001

   */


  try {


    const url =

      new URL(qrText);


    const id =

      url.searchParams.get("id");


    if (id) {

      return id.trim();

    }


  } catch (error) {


    /*

     * QR is not a URL.

     * Continue below.

     */


  }



  /*

   * If QR contains only:

   *

   * JBL-0001

   */


  if (qrText !== "") {

    return qrText;

  }



  return "";

}



/************************************************

 * RECORD ATTENDANCE

 ************************************************/


function recordAttendance(studentId) {


  showMessage(

    "Processing attendance...",

    "waiting"

  );



  /*

   * JSONP callback

   */


  const callbackName =

    "attendanceCallback_" +

    Date.now();



  window[callbackName] =

    function(response) {


      console.log(response);



      if (!response.success) {


        showMessage(

          "❌ " +

          response.message,

          "error"

        );


        cleanup();


        return;

      }



      /*

       * DUPLICATE

       */


      if (response.duplicate) {


        document.getElementById("result")

          .innerHTML =


          "<div class='student-name'>" +

          response.name +

          "</div>" +


          "<div class='student-info'>" +

          "Grade " +

          response.grade +

          " - " +

          response.section +

          "</div>" +


          "<div class='student-info'>" +

          "Time: " +

          response.time +

          "</div>" +


          "<div class='student-info'>" +

          "Status: " +

          response.status +

          "</div>" +


          "<div class='error'>" +

          "⚠ Attendance already recorded today" +

          "</div>";


        cleanup();


        return;

      }



      /*

       * SUCCESS

       */


      let statusClass =

        response.status === "PRESENT"

        ? "present"

        : "late";



      document.getElementById("result")

        .innerHTML =


        "<div class='student-name'>" +

        response.name +

        "</div>" +


        "<div class='student-info'>" +

        "Student ID: " +

        response.studentId +

        "</div>" +


        "<div class='student-info'>" +

        "Grade " +

        response.grade +

        " - " +

        response.section +

        "</div>" +


        "<div class='student-info'>" +

        "Time: " +

        response.time +

        "</div>" +


        "<div class='" +

        statusClass +

        "'>" +

        response.status +

        "</div>";



      /*

       * Wait before allowing next scan

       */


      setTimeout(function() {


        document.getElementById("result")

          .innerHTML =


          "<div class='waiting'>" +

          "Waiting for student QR code..." +

          "</div>";


      }, 4000);



      cleanup();


    };



  /*

   * Create script request

   */


  const script =

    document.createElement("script");



  const url =

    APPS_SCRIPT_URL +


    "?action=record" +


    "&id=" +

    encodeURIComponent(studentId) +


    "&station=" +

    encodeURIComponent(STATION_ID) +


    "&token=" +

    encodeURIComponent(STATION_TOKEN) +


    "&callback=" +

    encodeURIComponent(callbackName);



  script.src = url;



  script.onerror =

    function() {


      showMessage(

        "❌ Unable to connect to attendance server.",

        "error"

      );


      cleanup();


    };



  document.body.appendChild(script);



  function cleanup() {


    delete window[callbackName];


    if (script.parentNode) {

      script.parentNode.removeChild(script);

    }


  }


}



/************************************************

 * QR SCANNED

 ************************************************/


let processing = false;



function onScanSuccess(decodedText) {


  if (processing) {

    return;

  }



  processing = true;



  console.log(

    "QR scanned:",

    decodedText

  );



  const studentId =

    extractStudentID(decodedText);



  if (!studentId) {


    showMessage(

      "❌ Invalid student QR code.",

      "error"

    );


    setTimeout(function() {

      processing = false;

    }, 2000);


    return;

  }



  recordAttendance(studentId);



  /*

   * Allow next scan after processing

   */


  setTimeout(function() {


    processing = false;


  }, 5000);


}



/************************************************

 * QR SCANNER ERROR

 ************************************************/


function onScanFailure(error) {


  /*

   * Ignore normal scanning errors.

   *

   * The library continuously searches

   * for a QR code.

   */


}



/************************************************

 * START CAMERA

 ************************************************/


function startScanner() {


  const html5QrCode =

    new Html5Qrcode("reader");



  const config = {


    fps: 10,


    qrbox: {

      width: 250,

      height: 250

    }


  };



  html5QrCode.start(


    {

      facingMode: "environment"

    },


    config,


    onScanSuccess,


    onScanFailure


  )


  .catch(function(error) {


    console.error(error);



    showMessage(


      "❌ Camera could not start.<br><br>" +


      "Please allow camera permission " +

      "for this website.",


      "error"


    );


  });


}



/************************************************

 * START

 ************************************************/


window.addEventListener(

  "load",

  function() {


    startScanner();


  }

);
